async function readResponse(r){ const text = await r.text(); if(!text.trim()){ throw new Error(`Le serveur a renvoyé une réponse vide (HTTP ${r.status}).`); } let data; try{ data = JSON.parse(text); }catch(e){ throw new Error( `Réponse serveur invalide (HTTP ${r.status}).\n${text.slice(0,500)}` ); } if(!r.ok){ throw new Error(data.detail || data.error || `Erreur HTTP ${r.status}`); } return data; }
